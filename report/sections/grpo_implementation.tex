\problem{get\_response\_log\_probs}{Response log-probs and entropy}{1 point}

\problem{compute\_rollout\_rewards}{Computing rollout rewards}{1 point}

\deliverable{Implementation summary and the required validation result.}\answer

\problem{compute\_policy\_gradient\_loss\_on\_policy}{On-policy policy gradient}{1 point}

\problem{aggregate\_loss\_across\_microbatch\_sequence}{Aggregate loss across tokens and sequences}{0.5 points}
